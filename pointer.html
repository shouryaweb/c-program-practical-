POINTERS



1. Write a program to enter 5 values in an array using pointer and print them.

Code 
#include <stdio.h>

void main()
{
    int nums[5];
    int *ptr;
    int k;

    ptr = &nums[0];

    printf("Enter 5 values:\n");
    for (k = 0; k < 5; k++)
    {
        printf("Value %d: ", k + 1);
        scanf("%d", ptr);
        ptr++;
    }

    ptr = &nums[0];

    printf("\nValues entered are:\n");
    for (k = 0; k < 5; k++)
    {
        printf("%d\n", *ptr);
        ptr++;
    }
}
Output 
Enter 5 values:
Value 1: 10
Value 2: 20
Value 3: 30
Value 4: 40
Value 5: 50

Values entered are:
10
20
30
40
50

2. Write a program to enter values in a 2D array using a pointer and print the matrix.

Code 
#include <stdio.h>

void main()
{
    int mat[2][3];
    int *ptr;
    int k;

    ptr = &mat[0][0];

    printf("Enter 6 values for 2x3 matrix:\n");
    for (k = 0; k < 6; k++)
    {
        printf("Value %d: ", k + 1);
        scanf("%d", ptr);
        ptr++;
    }

    ptr = &mat[0][0];

    printf("\nMatrix is:\n");
    for (k = 0; k < 6; k++)
    {
        printf("%d\t", *ptr);
        ptr++;
        if ((k + 1) % 3 == 0)
        {
            printf("\n");
        }
    }
}
Output 
Enter 6 values for 2x3 matrix:
Value 1: 1
Value 2: 2
Value 3: 3
Value 4: 4
Value 5: 5
Value 6: 6

Matrix is:
1       2       3
4       5       6
