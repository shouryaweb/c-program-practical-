      18. String palindrome example (using reverse)
     

    

    

     

      Code
     

    

    
#include<stdio.h>
#include<conio.h>
#include<string.h>

void main()
{
    char word[50],rev[50];
    int k,n;
    clrscr();
    printf("Enter a string: ");
    fgets(word,sizeof(word),stdin);
    word[strcspn(word,"\n")]=0;
    n=strlen(word);
    for(k=0;k<n;k++)
    {
        rev[k]=word[n-1-k];
    }
    rev[n]='\0';
    if(strcmp(word,rev)==0)
    {
        printf("%s is a palindrome\n",word);
    }
    else
    {
        printf("%s is not a palindrome\n",word);
    }
}
    

    

     

      Output
     

    

    
Enter a string: madam
madam is a palindrome
    

   


   
   

    

     

      19. Enter the data from the user, the data consists of roll no., name, class, marks of 5 subjects
     

    

    

     

      Code
     

    

    
#include<stdio.h>
#include<conio.h>

void main()
{
    int rn,sub[5],k,sum=0;
    char sname[50],cls[20];
    float avg;
    clrscr();
    printf("Enter roll number: ");
    scanf("%d",&rn);

    printf("Enter name:");
    scanf("%s",sname);

    printf("Enter class:");
    scanf("%s",cls);

    printf("Enter marks of 5 subjects: \n");
    for(k=0;k<5;k++)
    {
        printf("subject %d:",k+1);
        scanf("%d",&sub[k]);
        sum+=sub[k];
    }
    avg=sum/5.0;

    printf("\n---Student Details---\n");
    printf("Roll No: %d\n",rn);
    printf("Name: %s\n",sname);
    printf("Class: %s\n",cls);
    printf("Total Marks: %d\n",sum);
    printf("Average Marks: %.2f\n",avg);
}
