3. Create structure of employee which there is emp.ID, name, designation of 50 employees from user, print name character with emp.ID in tabular form?

Code 
#include<stdio.h>
#include<conio.h>
#include<string.h>

struct employee
{
    int empid;
    char ename[50];
    char desig[50];
};
void main()
{
    struct employee e[50];
    int k, m, num;
    clrscr();
    printf("Enter number of employees(max 50): ");
    scanf("%d",&num);
    getchar();
    for(k=0;k<num;k++)
    {
        printf("Enter details of employee %d:\n",k+1);
        printf("Enter employee ID: ");
        scanf("%d",&e[k].empid);
        getchar();
        printf("Enter employee name: ");
        gets(e[k].ename);
        printf("Enter employee designation: ");
        gets(e[k].desig);
    }
    printf("\nEmp.Id\tName(character by character)\n");
    printf("------------------------------------------------\n");
    for(k=0;k<num;k++)
    {
        printf("%d\t",e[k].empid);
        for(m=0;e[k].ename[m]!='\0';m++)
        {
            printf("%c",e[k].ename[m]);
        }
        printf("\n");
    }
}

4. Enter student details like roll, name, fee, DOB and print all the detail by ascending order of roll number?

Code 
#include<stdio.h>
#include<conio.h>
#include<string.h>

struct student
{
    int rollno;
    char sname[50];
    int fees;
    char birth[15];
};
void main()
{
    struct student st[100], swap;
    int num, k, m;
    clrscr();
    printf("Enter number of students: ");
    scanf("%d",&num);

    for(k=0;k<num;k++)
    {
        printf("\nEnter details of student %d:\n",k+1);
        printf("Roll Number: ");
        scanf("%d",&st[k].rollno);
        printf("Name: ");
        scanf("%s",st[k].sname);
        printf("Fee: ");
        scanf("%d",&st[k].fees);
        printf("Date of birth(DD/MM/YYYY): ");
        scanf("%s",st[k].birth);
    }

    for(k=0;k<num;k++)
    {
        for(m=k+1;m<num;m++)
        {
            if(st[k].rollno>st[m].rollno)
            {
                swap=st[k];
                st[k]=st[m];
                st[m]=swap;
            }
        }
    }
    printf("\n Students details in Ascending order of roll number: \n");
    printf("-------------------------------------------------------------------\n");
    printf("Roll\tName\t\tFee\t\tDOB\n");
    printf("-------------------------------------------------------------------\n");
    for(k=0;k<num;k++)
    {
        printf("%d\t%-10s\t%d\t\t%s\n",st[k].rollno,st[k].sname,st[k].fees,st[k].birth);
    }
}
