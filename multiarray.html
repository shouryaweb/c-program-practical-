MULTI DIMENSIONAL ARRAY



1. Enter the four number from user and print it into matrix

Code 
#include <stdio.h>
void main() {
    int m[2][2], r, c;
    printf("Enter 4 numbers: ");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++){
            scanf("%d", &m[r][c]);
        }
    }
    printf("Matrix:\n");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++){
            printf("%d\t", m[r][c]);
        }
        printf("\n");
    }
}
Output 
Enter 4 numbers: 5 5 7 8
Matrix:
5       5
7       8

2. Enter the matrix from and transpose them ?

Code 
#include <stdio.h>
void main() {
    int orig[2][2], tr[2][2], x, y;
    printf("Enter 4 numbers for 2x2 matrix: ");
    for(x=0;x<2;x++){
        for(y=0;y<2;y++){
            scanf("%d",&orig[x][y]);
            tr[y][x] = orig[x][y];
        }
    }
    printf(" The Transpose Matrix is:\n");
    for(x=0;x<2;x++){
        for(y=0;y<2;y++){
            printf("%d\t ", tr[x][y]);
        }
        printf("\n");
    }
}
Output 
Enter 4 numbers for 2x2 matrix: 1 6 89 10
 The Transpose Matrix is:
1       89
6       10

4. Enter 2 matrix from user and subtract them ?

Code 
#include <stdio.h>
void main() {
    int p[2][2], q[2][2], diff[2][2], r, c;
    printf("Enter  numbers for  matrix-");
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            scanf("%d",&p[r][c]);
    printf("Enter  numbers for matrix- ");
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            scanf("%d",&q[r][c]);
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            diff[r][c] = p[r][c] - q[r][c];
    printf("Subtraction Matrix:\n");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++)
            printf("%d\t", diff[r][c]);
        printf("\n");
    }
}
