MULTI DIMENSIONAL ARRAY



1. Enter the four number from user and print it into matrix

Code 
#include <stdio.h>
void main() {
    int m[2][2], r, c;
    printf("Enter 4 numbers: ");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++){
            scanf("%d", &m[r][c]);
        }
    }
    printf("Matrix:\n");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++){
            printf("%d\t", m[r][c]);
        }
        printf("\n");
    }
}
Output 
Enter 4 numbers: 5 5 7 8
Matrix:
5       5
7       8

2. Enter the matrix from and transpose them ?

Code 
#include <stdio.h>
void main() {
    int orig[2][2], tr[2][2], x, y;
    printf("Enter 4 numbers for 2x2 matrix: ");
    for(x=0;x<2;x++){
        for(y=0;y<2;y++){
            scanf("%d",&orig[x][y]);
            tr[y][x] = orig[x][y];
        }
    }
    printf(" The Transpose Matrix is:\n");
    for(x=0;x<2;x++){
        for(y=0;y<2;y++){
            printf("%d\t ", tr[x][y]);
        }
        printf("\n");
    }
}
Output 
Enter 4 numbers for 2x2 matrix: 1 6 89 10
 The Transpose Matrix is:
1       89
6       10

3. Enter 2 matrix from user and add them ?

Code 
#include <stdio.h>
void main() {
    int first[2][2], second[2][2], total[2][2], r, c;
    printf("Enter  numbers for first matrix: ");
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            scanf("%d",&first[r][c]);
    printf("Enter  numbers for second matrix: ");
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            scanf("%d",&second[r][c]);
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            total[r][c] = first[r][c] + second[r][c];
    printf("Sum Matrix:\n");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++)
            printf("%d\t", total[r][c]);
        printf("\n");
    }
}
Output 
Enter  numbers for first matrix: 1 2 3 4
Enter  numbers for second matrix: 5 7 8 9
Sum Matrix:

5. Enter 2 matrix from user and multiply them ?

Code 
#include <stdio.h>
void main() {
    int m1[2][2], m2[2][2], prod[2][2], r, c, t;
    printf("Enter  numbers in matrix- ");
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            scanf("%d",&m1[r][c]);
    printf("Enter  numbers  matrix: ");
    for(r=0;r<2;r++)
        for(c=0;c<2;c++)
            scanf("%d",&m2[r][c]);
    for(r=0;r<2;r++){
        for(c=0;c<2;c++){
            prod[r][c]=0;
            for(t=0;t<2;t++)
                prod[r][c] += m1[r][t]*m2[t][c];
        }
    }
    printf("Multiplication Matrix\n");
    for(r=0;r<2;r++){
        for(c=0;c<2;c++)
            printf("%d\t", prod[r][c]);
        printf("\n");
    }
}
Output 
Enter  numbers in matrix- 1 2 3 4
Enter  numbers  matrix: 1 2 3 4
Multiplication Matrix
7       10
15      22
