FILE HANDLING



1. Create a file and copy that file into another file.

Code 
#include <stdio.h>
void main() {
    FILE *fp1, *fp2;
    char ch;
    fp1 = fopen("file1.txt", "w");
    fputs("Hello C Language", fp1);
    fclose(fp1);
    fp1 = fopen("file1.txt", "r");
    fp2 = fopen("file2.txt", "w");
    while((ch = fgetc(fp1)) != EOF) {
        fputc(ch, fp2);
    }
    fclose(fp1);
    fclose(fp2);
    printf("File copied successfully");
}
Output 
File copied successfully

3. Create two files and concatenate them into a 3rd file.

Code 
#include <stdio.h>
void main() {
    FILE *fp1, *fp2, *fp3;
    char ch;
    fp1 = fopen("file1.txt", "w");
    fputs("Hello ", fp1);
    fclose(fp1);
    fp2 = fopen("file2.txt", "w");
    fputs("World", fp2);
    fclose(fp2);
    fp1 = fopen("file1.txt", "r");
    fp2 = fopen("file2.txt", "r");
    fp3 = fopen("file3.txt", "w");
    while((ch = fgetc(fp1)) != EOF) {
        fputc(ch, fp3);
    }
    while((ch = fgetc(fp2)) != EOF) {
        fputc(ch, fp3);
    }
    fclose(fp1);
    fclose(fp2);
    fclose(fp3);
    printf("Files concatenated successfully");
}
Output 
Files concatenated successfully
(file3.txt contains: Hello World)

4. Create a file and count the number of characters in the file.

Code 
#include <stdio.h>
void main() {
    FILE *fp;
    char ch;
    int count = 0;
    fp = fopen("file1.txt", "w");
    fputs("Hello World", fp);
    fclose(fp);
    fp = fopen("file1.txt", "r");
    while((ch = fgetc(fp)) != EOF) {
        count++;
    }
    fclose(fp);
    printf("Number of characters = %d", count);
}
Output 
Number of characters = 11

5. Create a file and count the number of words in the file.

Code 
#include <stdio.h>
void main() {
    FILE *fp;
    char ch;
    int words = 0, inword = 0;
    fp = fopen("file1.txt", "w");
    fputs("I love C language", fp);
    fclose(fp);
    fp = fopen("file1.txt", "r");
    while((ch = fgetc(fp)) != EOF) {
        if(ch == ' ' || ch == '\n' || ch == '\t') {
            inword = 0;
        }
        else if(inword == 0) {
            inword = 1;
            words++;
        }
    }
    fclose(fp);
    printf("Number of words = %d", words);
}
Output 
Number of words = 4

6. Enter the file name from the user. Enter the data from the user and put the data in the file, but the data should be in capital letters.

Code 
#include <stdio.h>
#include <ctype.h>
void main() {
    char filename[100], data[100];
    int i;
    FILE *fp;
    printf("Enter file name: ");
    scanf("%s", filename);
    printf("Enter the data: ");
    scanf("%s", data);
    fp = fopen(filename, "w");
    for(i = 0; data[i] != '\0'; i++) {
        fputc(toupper(data[i]), fp);
    }
    fclose(fp);
    printf("Data written in capital letters");
}
Output 
Enter file name: test.txt
Enter the data: hello
Data written in capital letters
(test.txt contains: HELLO)

      Enter the file name from the user. Enter the data from the user till the user wants to enter the data. (Use gets)
     

    

    

     

      Code
     

    

    
#include <stdio.h>
void main() {
    char filename[100], data[100], ch;
    FILE *fp;
    printf("Enter file name: ");
    gets(filename);
    fp = fopen(filename, "w");
    do {
        printf("Enter the data: ");
        gets(data);
        fprintf(fp, "%s\n", data);
        printf("Do you want to enter more data? (y/n): ");
        ch = getchar();
        getchar();
    } while(ch == 'y' || ch == 'Y');
    fclose(fp);
    printf("Data saved in file");
}
